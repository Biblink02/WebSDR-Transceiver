cmake_minimum_required(VERSION 3.20)
project(websdr_dsp LANGUAGES C CXX)
include(FetchContent)
if(POLICY CMP0135)
    cmake_policy(SET CMP0135 NEW)
endif()
set(BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(BUILD_AUTOTESTS OFF CACHE BOOL "" FORCE)
set(BUILD_BENCHMARKS OFF CACHE BOOL "" FORCE)
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
set(BUILD_STATIC_LIBS ON CACHE BOOL "" FORCE)
set(FIND_FFTW OFF CACHE BOOL "" FORCE)
set(FIND_THREADS OFF CACHE BOOL "" FORCE)
set(ENABLE_SIMD OFF CACHE BOOL "" FORCE)
set(FIND_SIMD OFF CACHE BOOL "" FORCE)
set(ENABLE_LOGGING OFF CACHE BOOL "" FORCE)
set(ENABLE_TIMESTAMPS OFF CACHE BOOL "" FORCE)
if(NOT EMSCRIPTEN)
    set(CMAKE_POSITION_INDEPENDENT_CODE ON)
endif()
FetchContent_Declare(liquid
    URL https://codeload.github.com/jgaeddert/liquid-dsp/tar.gz/refs/tags/v1.8.3
    URL_HASH SHA256=18fa83b73db8bb6fe6ea0376e4b5aecf8645970f4604d10d9dadbf609f3f95e2)
FetchContent_MakeAvailable(liquid)

if(EMSCRIPTEN)
    add_executable(websdr_dsp src/receiver.cpp)
    set_target_properties(websdr_dsp PROPERTIES SUFFIX ".wasm")
    target_link_options(websdr_dsp PRIVATE --no-entry -sSTANDALONE_WASM=1
        -sALLOW_MEMORY_GROWTH=0 -sINITIAL_MEMORY=16777216 -sSTACK_SIZE=1048576
        -sFILESYSTEM=0 -sMALLOC=emmalloc -sSUPPORT_LONGJMP=0)
    target_compile_options(websdr_dsp PRIVATE -fno-exceptions)
else()
    add_library(websdr_channelizer SHARED src/channelizer.cpp)
    target_compile_features(websdr_channelizer PRIVATE cxx_std_20)
    target_compile_options(websdr_channelizer PRIVATE -Wall -Wextra -Wpedantic -Werror)
    target_include_directories(websdr_channelizer SYSTEM PRIVATE ${liquid_SOURCE_DIR}/include)
    target_link_libraries(websdr_channelizer PRIVATE liquid-static)
    add_library(websdr_dsp STATIC src/receiver.cpp)
    enable_testing()
    add_executable(receiver_test tests/receiver_test.cpp)
    target_link_libraries(receiver_test PRIVATE websdr_dsp)
    add_test(NAME receiver COMMAND receiver_test)
endif()
target_compile_features(websdr_dsp PUBLIC cxx_std_20)
target_compile_options(websdr_dsp PRIVATE -Wall -Wextra -Wpedantic -Werror)
target_include_directories(websdr_dsp PUBLIC src)
target_include_directories(websdr_dsp SYSTEM PRIVATE ${liquid_SOURCE_DIR}/include)
target_link_libraries(websdr_dsp PRIVATE liquid-static)
